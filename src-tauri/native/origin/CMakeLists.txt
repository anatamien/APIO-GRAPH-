cmake_minimum_required(VERSION 3.20)
project(apio_origin_helper VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /EHsc /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
include(CTest)
add_library(origin STATIC liborigin/OriginAnyParser.cpp liborigin/OriginParser.cpp)
target_include_directories(origin PUBLIC liborigin)
add_executable(origin-reader-tests reader-tests.cpp)
add_test(NAME origin-reader-tests COMMAND origin-reader-tests)
add_executable(origin-parser-tests parser-tests.cpp)
target_link_libraries(origin-parser-tests PRIVATE origin)
add_test(NAME origin-parser-tests COMMAND origin-parser-tests)
add_executable(apio-origin-helper decoder.cpp sandbox.cpp)
target_link_libraries(apio-origin-helper PRIVATE origin)
if(WIN32)
 target_link_libraries(apio-origin-helper PRIVATE userenv advapi32 ws2_32 ole32)
 if(MSVC)
  # The untrusted decoder does not need the broker's profile/COM libraries.
  target_link_libraries(apio-origin-helper PRIVATE delayimp)
  target_link_options(apio-origin-helper PRIVATE /DELAYLOAD:USERENV.dll /DELAYLOAD:ole32.dll /DELAYLOAD:WS2_32.dll /DELAYLOAD:ADVAPI32.dll)
 endif()
endif()
